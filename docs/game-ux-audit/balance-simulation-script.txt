function play(engine: any, setup: any, policy: (s: any) => any) {
  let s = engine.create(setup);
  let guard = 0;
  while (!engine.isOver(s) && guard++ < 300) {
    const a = policy(s);
    const why = engine.validate(s, a);
    if (why) throw new Error(`${engine.id}: ${why} :: ${JSON.stringify(a)}`);
    s = engine.apply(s, a);
  }
  return s;
}

  }

// defusal: a careful human that uses own + shared clues
const defCareful = (s: any) => {
  const p = def.defusalPrompt(s);
  if (p.phase === 'share') {
    if (!p.options.length) return { kind: 'share', clue: null };
    let bestK = p.options[0], bestN = Infinity;
    for (const k of p.options) { const n = def.consistent(s, [s.hands.player[k]]).length; if (n < bestN) { bestN = n; bestK = k; } }
    return { kind: 'share', clue: bestK };
  }
  const sh = def.sharedClues(s);
  const clues = [...s.hands.player, ...[...sh.bot, ...(p.shared !== null ? [p.shared] : [])].map((k: number) => s.hands.bot[k])];
  const safe = def.safety(s, def.consistent(s, clues));
  let w = -1, b = -1;
  safe.forEach((v: any, i: number) => { if (v !== null && v > b) { b = v; w = i; } });
  return { kind: 'cut', wire: w };
};

const dP = (bid: (s: any, prev: any, own: number[]) => any, call: (s: any, prev: any, own: number[]) => boolean) => (s: any) => {
  const bids = dice.liveBids(s); const prev = bids[bids.length - 1]; const own = dice.roll(s).player;

};

const cbGuess = (p: any, stopAfter: number) => {
  if (p.guesses.length >= Math.min(stopAfter, p.clue.n)) return { kind: 'stop' };
  const open = p.board.words.map((w: string, i: number) => ({ i, l: code.link(w, p.clue.theme) })).filter((x: any) => !p.guesses.includes(x.i)).sort((a: any, b: any) => b.l - a.l);
  return { kind: 'guess', word: open[0].i };
};
const cbClue = (p: any) => {
  let bt = code.THEMES[0], bv = -99, bn = 1;
  for (const t of code.THEMES) {
    const ts = p.board.words.map((w: string, i: number) => ({ k: p.board.key[i], l: code.link(w, t) }));
    const hit = ts.filter((x: any) => x.k === 'target' && x.l >= 2).length;
    const bad = ts.filter((x: any) => x.k === 'trap' && x.l >= 1).length;
    const neu = ts.filter((x: any) => x.k === 'neutral' && x.l >= 2).length;
    const v = hit - 3 * bad - neu;
    if (v > bv) { bv = v; bt = t; bn = Math.max(1, Math.min(3, hit)); }
  }
  return { kind: 'clue', theme: bt, n: bn };
};

function classic(id: string, policies: Record<string, (h: any[], p: any) => any>) {
  const e = ENGINES[id];
  console.log(`\n## classic ${id} (${e.rounds} rounds; ${N} seeds; win/draw/loss %)`);
  for (const [name, pol] of Object.entries(policies)) {
    const parts: string[] = [];
    for (const difficulty of ['easy', 'medium', 'hard']) {
      let w = 0, d = 0, l = 0, pts = 0, c = 0; const per: string[] = [];
      for (const bot of e.bots) {
        let pw = 0;
        for (let i = 0; i < N; i++) {
          const spec = { bot, difficulty, seed: (i * 2654435761 + 7) >>> 0 };
          const h: any[] = [];
          while (h.length < e.rounds) { const p = e.prompt(spec, h); h.push(e.resolve(spec, h, e.parseMove(pol(h, p), p))); }
          const o = e.outcome(h); if (o === 'win') { w++; pw++; } else if (o === 'draw') d++; else l++;
          pts += e.totals(h).player; c++;
        }
        per.push(`${bot.slice(0, 5)}:${Math.round((pw / N) * 100)}`);
      }
      parts.push(`${difficulty[0].toUpperCase()} ${Math.round((w / c) * 100)}/${Math.round((d / c) * 100)}/${Math.round((l / c) * 100)} [${per.join(' ')}] pts=${(pts / c).toFixed(1)}`);
    }
    console.log(`- ${name}: ` + parts.join(' | '));
  }
}
